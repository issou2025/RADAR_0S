#!/bin/bash

# CLEAN HANDS CLEAN MONEY FAM - PRE-COMMIT IP PROTECTION HOOK
# Owner: Morley Moses Apooch
# Purpose: Enforce copyright headers and generate proof-of-creation hashes.

echo "🔒 Running IP Protection Validation..."

# Define required header snippet
REQUIRED_HEADER="Morley Moses Apooch"
LAYER1_DIR="01_Core_Source"
MANIFEST_FILE="Compliance/IP_Protection/SHA256_MANIFEST.txt"

# Ensure manifest directory exists
mkdir -p $(dirname $MANIFEST_FILE)

# Check for missing headers in tracked files
MISSING_HEADERS=0
for file in $(git diff --cached --name-only --diff-filter=ACM); do
    # Skip non-code files and directories
    if [[ "$file" == *.py || "$file" == *.dart || "$file" == *.jsx || "$file" == *.js || "$file" == *.sh ]]; then
        if ! grep -q "$REQUIRED_HEADER" "$file"; then
            echo "❌ ERROR: Missing copyright header in $file"
            echo "   Add: 'Copyright (c) 2026 Morley Moses Apooch'"
            MISSING_HEADERS=1
        fi
    fi
done

if [ $MISSING_HEADERS -ne 0 ]; then
    echo "❌ Commit rejected. Fix copyright headers."
    exit 1
fi

# Generate SHA-256 for Layer 1 files (Proof of Creation)
if [ -d "$LAYER1_DIR" ]; then
    echo "Generating SHA-256 Manifest for Layer 1..."
    > $MANIFEST_FILE
    echo "Layer 1 SHA-256 Proof-of-Creation Manifest" >> $MANIFEST_FILE
    echo "Generated: $(date -u +"%Y-%m-%d %H:%M:%S UTC")" >> $MANIFEST_FILE
    echo "" >> $MANIFEST_FILE
    
    while IFS= read -r -d '' file; do
        HASH=$(sha256sum "$file" | awk '{print $1}')
        echo "$HASH  $file" >> $MANIFEST_FILE
    done < <(find "$LAYER1_DIR" -type f \( -name "*.py" -o -name "*.dart" -o -name "*.jsx" \) -print0)
    
    echo "✅ SHA-256 Manifest updated: $MANIFEST_FILE"
else
    echo "⚠️  Layer 1 directory not found. Skipping hash generation."
fi

echo "✅ IP Protection Validation Passed."
exit 0
