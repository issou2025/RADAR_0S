# Project Proposal — Flutter App

## Project Understanding
You need assistance with: 
**[Windows] Focusing a nested popup closes both popups and sends the main window behind other applications**

*Project description:*
This issue occurs on Windows, but not on Linux. I haven't tested on other platforms.

### Steps to reproduce

1. Create a new Flutter project, paste the code sample below into `main.dart`, and run with `-d windows`
2. Click to open the first popup, and then click the button to focus it - this works as intended
3. Click the button to open the second popup, then click to focus it

### Expected results

The popups both remain open, and the main window remains on top.

### Actual results

The popups both close, and the window is sent behind other open windows.

### Code sample

<details><summary>Code sample</summary>

```dart
import 'package:flutter/widgets.dart';
import 'package:flutter/src/foundation/_features.dart' as experimental;
import 'package:flutter/src/widgets/_window.dart';
import 'package:flutter/src/widgets/_window_positioner.dart';

void main() {
  experimental.isWindowingEnabled = true;
  WidgetsFlutterBinding.ensureInitialized();
  final window = _RunnerWindow();
  final popups = ValueNotifier<List<PopupWindowController>>([]);

  void openPopup() {
    if (popups.value.length == 2) return;
    late final PopupWindowController popup;
    popup = PopupWindowController(
      // The first popup belongs to the runner window; the second to the first.
      parent: popups.value.isEmpty ? window : popups.value.last,
      anchorRect: popups.value.isEmpty
          ? const Rect.fromLTWH(40, 80, 1, 1)
          : const Rect.fromLTWH(280, 0, 1, 1),
      positioner: const WindowPositioner(
        parentAnchor: WindowPositionerAnchor.topLeft,
        childAnchor: WindowPositionerAnchor.topLeft,
      ),
      delegate: _PopupDelegate(() {
        popups.value = popups.value.where((p) => p != popup).toList();
      }),
    );
    popups.value = [...popups.value, popup];
  }

  Widget button(String label, VoidCallback onTap) => GestureDetector(
    onTap: onTap,
    child: Container(
      margin: const EdgeInsets.only(top: 12),
      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 8),
      decoration: BoxDecoration(
        color: const Color(0xffffffff),
        border: Border.all(),
      ),
      child: Text(label),
    ),
  );

  Widget popupPanel(
    String title,
    Color color, {
    VoidCallback? openSecond,
  }) => Focus(
    child: Builder(
      builder: (context) => ColoredBox(
        color: color,
        child: SizedBox(
          width: 280,
          child: Padding(
            padding: const EdgeInsets.all(12),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Text(title),
                Text(
                  Focus.of(context).hasFocus
                      ? 'Flutter focus: focused'
                      : 'Flutter focus: not focused',
                ),
                // Request focus in this popup only when its button is pressed.
                button(
                  'Focus this popup',
                  () => Focus.of(context).requestFocus(),
                ),
                if (openSecond != null) button('Open second popup', openSecond),
              ],
            ),
          ),
        ),
      ),
    ),
  );

  runApp(
    Directionality(
      textDirection: TextDirection.ltr,
      child: DefaultTextStyle(
        style: const TextStyle(color: Color(0xff000000), fontSize: 18),
        child: ValueListenableBuilder(
          valueListenable: popups,
          builder: (context, controllers, _) => ViewAnchor(
            // Each popup renders into its own native view.
            view: ViewCollection(
              views: [
                for (final (index, controller) in controllers.indexed)
                  PopupWindow(
                    key: ObjectKey(controller),
                    controller: controller,
                    child: popupPanel(
                      index == 0 ? 'First popup' : 'Second popup',
                      index == 0
                          ? const Color(0xffc8f2cc)
                          : const Color(0xffffd0b3),
                      openSecond: index == 0 ? openPopup : null,
                    ),
                  ),
              ],
            ),
            child: ColoredBox(
              color: const Color(0xffc8e6ff),
              child: Center(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    const Text('Main window'),
                    button('Open first popup', () {
                      if (popups.value.isEmpty) openPopup();
                    }),
                  ],
                ),
              ),
            ),
          ),
        ),
      ),
    ),
  );
}

// Only rootView is needed to use the standard runner window as a popup parent.
class _RunnerWindow extends RegularWindowController {
  _RunnerWindow() : super.empty() {
    rootView = WidgetsBinding.instance.platformDispatcher.implicitView!;
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

class _PopupDelegate extends PopupWindowControllerDelegate {
  _PopupDelegate(this.onDestroyed);
  final VoidCallback onDestroyed;

  @override
  void onWindowDestroyed() => onDestroyed();
}

```
