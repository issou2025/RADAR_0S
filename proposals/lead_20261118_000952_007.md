## The product and why
**Freightline**, a multi-tenant freight-forwarding platform. Each tenant is a forwarding company; its staff and its customers log in to the same system.
- Sales quotes a multi-leg move with a hidden margin.
- An accepted quote becomes a shipment whose legs are booked with outside carriers through their API, and a failed booking must undo the legs already booked.
- Customs can hold a shipment.
- Delivery is billed by a separate finance backend, and customers file claims with evidence files.

I chose it because nearly every aggregate points into another bounded context, the main process is a saga with compensation against an outside system, and it needs money, deadlines, files, row-level customer isolation and two backends talking over a broker. That stresses the places where a DSL usually breaks.

Earlier evals mostly got stuck on generated code that didn't compile or crashed at runtime. The friction has moved up a level: from "the generated code is wrong" to "the language can't say what a large domain needs". ## Verdict as the architect
**Would I adopt it today for Freightline? No.**

**For what? Yes, for a single-backend product, or one whose contexts don't need strict cross-context references:** an internal operations tool, or one bounded context of a larger estate (for example the finance service on its own). It is especially worth it where the multi-target matrix, tenancy, masking and audit would otherwise be hand-written.

**Three things that would change my answer:**
1. **Cross-context references that work with deny-by-default tenancy** (#6274, data-boundaries design #5261). Today I must choose between modelling my bounded contexts honestly and keeping the security default.
2. **A saga that can talk to an outside system:** typed external API responses (#3358), per-item steps over its input (#6275), an explicit failure and compensation path (#6185), and a failed reaction that surfaces to the caller or retries (#5400).
3. **One frontend over several backends** (#3333 / #6136), so a split backend doesn't force a split or missing UI.

And as a baseline: a fresh `generate system` that compiles on every chosen target with zero hand patches.

// Freightline — a multi-tenant freight-forwarding platform.
// Each tenant is a forwarding company (Forwarder).  Sales quotes a customer a
// multi-leg move; an accepted quote becomes a Shipment whose legs are booked
// with carriers through an external carrier API; customs can hold a shipment;
// delivery raises an event that the Finance backend (another deployable, over
// a broker) turns into an invoice.  Customers file claims with evidence files.

system Freightline {

  user {
    id: string
    email: string
    tenantId: string
    role: string
    permissions: string[]
    customerId: Customer id?
  }

  tenancy by user.tenantId of Forwarder

  auth {
    enforcement: opt
    provider: keycloak
    oidc {
      issuer: env("OIDC_ISSUER")
      audience: env("OIDC_AUDIENCE")
      clientId: env("OIDC_CLIENT_ID")
    }
  }

  i18n { source: "en" }

  subdomain Platform {
    context Tenants {
      policy IsPlatformAdmin(): bool = currentUser.role == "platformAdmin"
      aggregate Forwarder with crudish(requires: IsPlatformAdmin) {
        name: string
        baseCurrency: string
        invariant baseCurrency.length == 3
      }
      repository Forwarders for Forwarder { }
      policy { read Forwarder requires IsPlatformAdmin() }
    }
  }

  subdomain Sales {
    permissions { quoteApprove, marginView, customerManage }

    context Quoting {
      enum TransportMode { Ocean, Air, Road, Rail }
      enum QuoteStatus { Draft, Sent, Accepted, Rejected, Expired }

      valueobject Location {
        unlocode: string
        name: string
        invariant unlocode.length == 5
      }

      aggregate Customer with tenantOwned, auditable {
        name: string
        derived display: string = name
        creditLimit: money
        create(name: string, creditLimit: money) {
          requires currentUser.permissions.contains(permissions.customerManage)
        }
      }
      repository Customers for Customer { }

      aggregate Quote with tenantOwned, auditable {
        reference: string
        derived display: string = reference
        customer: Customer id
        origin: Location
        destination: Location
        validUntil: datetime managed
        stamp onCreate { validUntil := now() + days(14) }
        status: QuoteStatus managed = Draft
        margin: money mask unless currentUser.permissions.contains(permissions.marginView)
        lines: QuoteLine[]

        entity QuoteLine {
          mode: TransportMode
          pickup: Location
          dropoff: Location
          price: money
          transitDays: int
          invariant transitDays > 0
        }

        derived total: money = lines.sum(l => l.price) + margin

        invariant lines.count > 0 || status == Draft

        create(reference: string, customer: Customer id, origin: Location, destination: Location, margin: money) {
          requires currentUser.role == "sales"
        }

        operation addLine(mode: TransportMode, pickup: Location, dropoff: Location, price: money, transitDays: int) requires currentUser.role == "sales" when status == Draft {
          lines += QuoteLine { mode: mode, pickup: pickup, dropoff: dropoff, price: price, transitDays: transitDays }
        }

        operation send() requires currentUser.role == "sales" when status == Draft {
          precondition lines.count > 0
          status := Sent
          emit QuoteSent { quote: id, customer: customer }
        }

        operation accept() requires currentUser.permissions.contains(permissions.quoteApprove) when status == Sent {
          precondition now() < validUntil
          status := Accepted
          emit QuoteAccepted { quote: id, customer: customer, total: total, legs: lines.map(l => LegPlan { mode: l.mode, etd: now(), eta: now() + days(l.transitDays) }) }
        }

        operation expire() requires currentUser.isSystem when status == Sent {
          status := Expired
        }
      }
      repository Quotes for Quote { }

      event QuoteSent { quote: Quote id, customer: Customer id }
      valueobject LegPlan {
        mode: TransportMode
        etd: datetime
        eta: datetime
      }
      event QuoteAccepted { quote: Quote id, customer: Customer id, total: money, legs: LegPlan[] }

      policy {
        read Customer requires currentUser.role == "sales" || currentUser.role == "ops"
        read Quote requires true
        allow read on Quote as q where q.customer == currentUser.customerId
      }
    }
  }

  subdomain Operations {
    permissions { customsRelease, carrierManage }

    context Shipping {
      policy CanManageCarriers(): bool = currentUser.permissions.contains(permissions.carrierManage)
      enum ShipmentStatus { Planned, Booked, InTransit, OnHold, Delivered, Cancelled }
      enum LegStatus { Pending, Reserved, Failed, Cancelled }
      enum HoldReason { Documents, Inspection, Duties }

      aggregate Carrier with tenantOwned, crudish(requires: CanManageCarriers) {
        name: string
        derived display: string = name
        scac: string
        invariant scac.length == 4
      }
      repository Carriers for Carrier { }

      aggregate Shipment with tenantOwned, auditable {
        quoteRef: Quote id
        customerRef: Customer id
        status: ShipmentStatus managed = Planned
        billOfLading: File?
        legs: Leg[]
        holds: CustomsHold[]
        deliveredAt: datetime?

        entity Leg {
          seq: int
          carrier: Carrier id?
          mode: TransportMode
          etd: datetime
          eta: datetime
          bookingRef: string?
          legStatus: LegStatus
          invariant eta > etd
        }

        entity CustomsHold {
          reason: HoldReason
          placedAt: datetime
          releasedAt: datetime?
        }

        derived openHolds: int = holds.where(h => h.releasedAt == null).count

        operation markBooked() requires currentUser.isSystem when status == Planned {
          precondition legs.all(l => l.legStatus == Reserved)
          status := Booked
        }

        operation depart() requires currentUser.role == "ops" when status == Booked {
          status := InTransit
          emit ShipmentDeparted { shipment: id, customer: customerRef }
        }

        operation placeHold(reason: HoldReason) requires currentUser.role == "ops" when status == InTransit {
          holds += CustomsHold { reason: reason, placedAt: now() }
          status := OnHold
          emit ShipmentHeld { shipment: id, customer: customerRef, reason: reason }
        }

        operation releaseHolds() requires currentUser.permissions.contains(permissions.customsRelease) when status == OnHold {
          holds := holds.map(h => CustomsHold { reason: h.reason, placedAt: h.placedAt, releasedAt: now() })
          status := InTransit
        }

        operation deliver(amount: money) requires currentUser.role == "ops" when status == InTransit {
          status := Delivered
          deliveredAt := now()
          emit ShipmentDelivered { shipment: id, customer: customerRef, amount: amount, at: now() }
        }

        operation cancel() requires currentUser.role == "ops" || currentUser.isSystem when status == Planned || status == Booked {
          status := Cancelled
        }
      }
      repository Shipments for Shipment { }
      policy {
        read Carrier requires currentUser.role == "ops"
        read Shipment requires currentUser.role == "ops"
      }

      event ShipmentDeparted { shipment: Shipment id, customer: Customer id }
      event ShipmentHeld { shipment: Shipment id, customer: Customer id, reason: HoldReason }
      event ShipmentDelivered { shipment: Shipment id, customer: Customer id, amount: money, at: datetime }

      channel Deliveries { carries: ShipmentDelivered  delivery: queue  retention: work }

      // Wanted: reserve every leg with the carrier API and cancel the earlier
      // reservations when one fails. Expressible: one POST, then book.
      workflow BookShipment requires currentUser.role == "ops" {
        quoteId: Quote id
        create(e: QuoteAccepted) by e.quote {
          let s = Shipment.create(quoteRef: e.quote, customerRef: e.customer)
          carrierApi.post("/reservations", { shipment: s.id, legs: e.legs.count })
          s.markBooked()
        }
      }

      projection CustomerBoard keyed by customer requires currentUser.role == "ops" {
        customer: Customer id
        inTransit: int
        onHold: int
        delivered: int
        on(e: ShipmentDeparted) { inTransit := inTransit + 1 }
        on(e: ShipmentHeld) { onHold := onHold + 1 }
        on(e: ShipmentDelivered) { delivered := delivered + 1  inTransit := inTransit - 1 }
      }
    }
  }

  subdomain Finance {
    permissions { claimApprove, invoiceVoid }

    context Billing {
      enum InvoiceStatus { Issued, Paid, Void }
      enum ClaimStatus { Filed, Approved, Rejected }

      aggregate Invoice with tenantOwned, auditable {
        shipment: Shipment id
        customer: Customer id
        amount: money
        dueAt: datetime
        status: InvoiceStatus managed = Issued

        operation pay() requires currentUser.role == "finance" when status == Issued {
          status := Paid
        }
        operation void() requires currentUser.permissions.contains(permissions.invoiceVoid) when status == Issued {
          status := Void
        }
      }
      repository Invoices for Invoice { }

      aggregate Claim with tenantOwned, auditable {
        shipment: Shipment id
        amount: money
        description: string
        evidence: File?
        status: ClaimStatus managed = Filed

        create(shipment: Shipment id, amount: money, description: string, evidence: File?) {
          requires currentUser.customerId != null
        }
        operation approve() requires currentUser.permissions.contains(permissions.claimApprove) when status == Filed {
          status := Approved
        }
        operation reject() requires currentUser.permissions.contains(permissions.claimApprove) when status == Filed {
          status := Rejected
        }
      }
      repository Claims for Claim { }
      policy {
        read Invoice requires currentUser.role == "finance"
        read Claim requires currentUser.role == "finance"
      }

      workflow InvoiceOnDelivery requires currentUser.role == "finance" {
        shipmentId: Shipment id
        create(e: ShipmentDelivered) by e.shipment {
          let inv = Invoice.create(shipment: e.shipment, customer: e.customer, amount: e.amount, dueAt: now() + days(30))
        }
      }
    }
  }

  api SalesApi from Sales
  api OperationsApi from Operations
  api FinanceApi from Finance
  api PlatformApi from Platform

  ui OpsPortal with scaffold(subdomains: [Sales, Operations, Platform]) { }
  ui CustomerPortal with scaffold(subdomains: [Sales]) {
    framework: vue
  }

  storage opsDb { type: postgres }
  storage financeDb { type: postgres }
  storage bus { type: rabbitmq }
  storage docs { type: localDisk }
  storage carrierGateway { type: restApi, config: { baseUrl: "https://carriers.example.com" } }

  resource tenantsState { for: Tenants, kind: state, use: opsDb }
  resource quotingState { for: Quoting, kind: state, use: opsDb }
  resource shippingState { for: Shipping, kind: state, use: opsDb }
  resource shippingFiles { for: Shipping, kind: objectStore, use: docs }
  resource carrierApi { for: Shipping, kind: api, use: carrierGateway }
  resource billingState { for: Billing, kind: state, use: financeDb }
  resource billingFiles { for: Billing, kind: objectStore, use: docs }

  channelSource deliveriesBus { for: Deliveries, use: bus }

  deployable ops {
    platform: node
    contexts: [Tenants, Quoting, Shipping]
    dataSources: [tenantsState, quotingState, shippingState, shippingFiles, carrierApi]
    channels: [deliveriesBus]
    serves: SalesApi, OperationsApi, PlatformApi
    port: 3000

  }
