# Project Proposal — Flutter App

## Project Understanding
You need assistance with: 
**[iOS] Blank block over the top page during a route transition when platform views on both routes overlap**

*Project description:*
### Steps to reproduce

1. Create a new app with `flutter create --platforms ios`.
2. Replace `ios/Runner/AppDelegate.swift` and `lib/main.dart` with the code below. The only native code is a factory for a plain `UIButton` (`.glass()` configuration on iOS 26+, so it has the interactive glass press), shown through `UiKitView` with an `EagerGestureRecognizer`; taps come back over a method channel. No third-party packages.
3. Run on an iOS 26+ device or simulator (reproduces on both).
4. Tap **+** (a `UiKitView` in the FAB slot) a few times to push pages. Every page has `UiKitView` buttons in its `AppBar` and FAB, in the same positions.
5. Go back with a slow edge swipe: drag halfway, hold, let go, repeat. Also go back with the back button (top left).

### Expected results

The top page slides away over the page underneath, with all of its Flutter content visible.

### Actual results

During the transition a blank rectangle covers part of the top page's `AppBar`. The Flutter content inside it is missing, while the platform views stay visible. It goes away when the transition ends.

- The rectangle is filled with the **top page's own background colour**. With "Alternate page colours" on (even pages yellow, odd pages blue), the rectangle on a blue page is blue, not the yellow of the page underneath.
- It sits where the **underlying page's** platform views are at that moment (shifted by the parallax), not where the top page's own buttons are.
- It is the same colour and in the same place whether the platform views are glass or solid red, so it isn't the underlying platform view showing through. The top page's Flutter content is missing in that area.
- It is the same whether the `UIButton`s take touches themselves (interactive glass) or have user interaction disabled.

Knobs on the Home page, and whether the block still appears with each one off. None of them is needed: one platform view at the same spot on two stacked `MaterialPageRoute`s in a plain `MaterialApp` is enough.

| Knob off | Block still appears? |
|---|---|
| Nested Navigator | Yes |
| Body behind AppBar | Yes |
| Native buttons in AppBar (FAB only) | Yes, around the FAB (smaller and harder to see than in the `AppBar`) |
| Native FAB (AppBar only) | Yes |

```dart
import 'package:flutter/foundation.dart';
import 'package:flutter/gestures.dart';
import 'package:flutter/services.dart';
import 'package:material_ui/material_ui.dart';

final nestedNavigator = ValueNotifier(true);
final bodyBehindAppBar = ValueNotifier(true);
final appBarButtons = ValueNotifier(true);
final fabButton = ValueNotifier(true);
final solidButtons = ValueNotifier(false);
final alternateColours = ValueNotifier(false);

Listenable get knobs => Listenable.merge([
  alternateColours,
  nestedNavigator,
  bodyBehindAppBar,
  appBarButtons,
  fabButton,
  solidButtons,
]);

void main() => runApp(const ReproApp());

class ReproApp extends StatelessWidget {
  const ReproApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(colorSchemeSeed: const Color(0xFF0B5A2E)),
      home: ValueListenableBuilder<bool>(
        valueListenable: nestedNavigator,
        builder: (context, nested, _) => nested
            ? Navigator(
                key: const ValueKey('nested'),
                onGenerateRoute: (_) => MaterialPageRoute<void>(
                  builder: (_) => const ReproPage(depth: 0),
                ),
              )
            : const ReproPage(depth: 0),
      ),
    );
  }
}

class NativeButton extends StatelessWidget {
  const NativeButton({
    super.key,
    required this.symbol,
    this.size = 40,
    this.onTap,
  });

  final String symbol;
  final double size;
  final VoidCallback? onTap;

  @override
  Widget build(BuildContext context) {
    return SizedBox.square(
      dimension: size,
      child: UiKitView(
        key: ValueKey(solidButtons.value),
        viewType: 'plain-button',
        creationParams: <String, Object>{
          'symbol': symbol,
          'solid': solidButtons.value,
        },
        creationParamsCodec: const StandardMessageCodec(),
        gestureRecognizers: <Factory<OneSequenceGestureRecognizer>>{
          Factory<EagerGestureRecognizer>(() => EagerGestureRecognizer()),
        },
        onPlatformViewCreated: (id) {
          MethodChannel('plain-button/$id').setMethodCallHandler((call) async {
            if (call.method == 'tap') onTap?.call();
          });
        },
      ),
    );
  }
}

class ReproPage extends StatelessWidget {
  const ReproPage({super.key, required this.depth});

  final int depth;

  void _push(BuildContext context) {
    Navigator.of(context).push(
      MaterialPageRoute<void>(builder: (_) => ReproPage(depth: depth + 1)),
    );
  }

  Widget _knob(String label, ValueNotifier<bool> value) {
    return SwitchListTile(
      title: Text(label),
      value: value.value,
      onChanged: (v) => value.value = v,
    );
  }

  @override
  Widget build(BuildContext context) {
    return ListenableBuilder(
      listenable: knobs,
      builder: (context, _) {
        final behind = bodyBehindAppBar.value;
        final page = !alternateColours.value
            ? const Color(0xFFF2F2F7)
            : depth.isEven
            ? const Color(0xFFFFF3C4)
            : const Color(0xFFCDE7FF);
        final top = behind
            ? MediaQuery.paddingOf(context).top + kToolbarHeight + 16
            : 16.0;
        return Scaffold(
          backgroundColor: page,
          extendBodyBehindAppBar: behind,
          appBar: AppBar(
            backgroundColor: page.withValues(alpha: 0.8),
            surfaceTintColor: Colors.transparent,
            scrolledUnderElevation: 0,
            automaticallyImplyLeading: false,
            leadingWidth: 64,
            leading: depth == 0 || !appBarButtons.value
                ? null
                : Center(
                    child: NativeButton(
                      symbol: 'chevron.left',
                      onTap: () => Navigator.of(context).maybePop(),
                    ),
                  ),
            title: Text(depth == 0 ? 'Home' : 'Page $depth'),
            actions: appBarButtons.value
                ? const [
                    NativeButton(symbol: 'bell'),
                    SizedBox(width: 8),
                    NativeButton(symbol: 'pencil'),
                    SizedBox(width: 16),
                  ]
                : null,
          ),
          body: ListView(
            padding: EdgeInsets.fromLTRB(16, top, 16, 140),
            children: [
              if (depth == 0) ...[
                Card(
                  child: Column(
                    children: [
                      _knob('Nested Navigator', nestedNavigator),
                      _knob('Body behind AppBar', bodyBehindAppBar),
                      _knob('Native buttons in AppBar', appBarButtons),
                      _knob('Native FAB', fabButton),
                      _knob('Solid red buttons', solidButtons),
                      _knob('Alternate page colours', alternateColours),
                    ],
                  ),
                ),
                const Padding(
                  padding: EdgeInsets.all(8),
                  child: Text(
                    'Every round button is a plain UIKit UIButton in a '
                    'UiKitView. Push a few pages with the + button, then go '
                    'back with a slow edge swipe (hold halfway, let go, '
                    'repeat) and with the back button. Watch for a blank '
                    'block over the top page.',
                  ),
                ),
              ],
              FilledButton(
                onPressed: () => _push(context),
                child: Text('Push page ${depth + 1}'),
              ),
              const SizedBox(height: 12),
              for (var i = 1; i <= 16; i++)
                Card(
                  child: ListTile(
                    title: Text(
                      'Row $i on ${depth == 0 ? 'Home' : 'page $depth'}',
                    ),
                    subtitle: const Text('Some text so the page has content'),
                    onTap: () => _push(context),
                  ),
                ),
            ],
          ),
          floatingActionButton: fabButton.value
              ? NativeButton(
                  symbol: 'plus',
                  size: 56,
                  onTap: () => _push(context),
                )
              : null,
        );
      },
    );
  }
}
```

```swift
import Flutter
import UIKit

@main
@objc class AppDelegate: FlutterAppDelegate, FlutterImplicitEngineDelegate {
  override func application(
    _ application: UIApplication,
    didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?
  ) -> Bool {
    return super.application(application, didFinishLaunchingWithOptions: launchOptions)
  }

  func didInitializeImplicitFlutterEngine(_ engineBridge: FlutterImplicitEngineBridge) {
    GeneratedPluginRegistrant.register(with: engineBridge.pluginRegistry)
    if let registrar = engineBridge.pluginRegistry.registrar(forPlugin: "PlainButtonFactory") {
      registrar.register(
        PlainButtonFactory(messenger: registrar.messenger()),
        withId: "plain-button"
      )
    }
  }
}

final class PlainButtonFactory: NSObject, FlutterPlatformViewFactory {
  private let messenger: FlutterBinaryMessenger

  init(messenger: FlutterBinaryMessenger) {
    self.messenger = messenger
    super.init()
  }

  func create(withFrame frame: CGRect, viewIdentifier viewId: Int64, arguments args: Any?) -> FlutterPlatformView {
    let params = args as? [String: Any]
    let symbol = params?["symbol"] as? String ?? "circle"
    let solid = params?["solid"] as? Bool ?? false
    let channel = FlutterMethodChannel(name: "plain-button/\(viewId)", binaryMessenger: messenger)
    return PlainButtonView(frame: frame, symbol: symbol, solid: solid, channel: channel)
  }

  func createArgsCodec() -> FlutterMessageCodec & NSObjectProtocol {
    FlutterStandardMessageCodec.sharedInstance()
  }
}

final class PlainButtonView: NSObject, FlutterPlatformView {
  private let button: UIButton

  init(frame: CGRect, symbol: String, solid: Bool, channel: FlutterMethodChannel) {
    var config: UIButton.Configuration
    if solid {
      config = .filled()
      config.baseBackgroundColor = .systemRed
      config.baseForegroundColor = .white
    } else if #available(iOS 26.0, *) {
      config = .glass()
    } else {
      config = .filled()
    }
    config.image = UIImage(systemName: symbol)
    config.cornerStyle = .capsule
    button = UIButton(configuration: config)
    button.frame = frame
    button.addAction(
      UIAction { _ in channel.invokeMethod("tap", arguments: nil) },
      for: .primaryActionTriggered
    )
    super.init()
  }

  func view() -> UIView { button }
}
```
